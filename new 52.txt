namespace Drupal\highwire_markup\Plugin\HWMarkup\Processor;

use Drupal\highwire_markup\ProcessorBase;
use HighWire\Parser\Markup\Markup;

/**
 * Brightcove Video Processor.
 *
 * @Processor(
 *   id = "brightcove_video",
 *   name = "Brightcove Video",
 *   description = "Process Brightcove video markup"
 * )
 */
class BrightcoveVideo extends ProcessorBase {

  /**
   * {@inheritdoc}
   */
  public function process(Markup $markup, &$context = NULL) {

    $config = $this->getConfig();

    $account = $config['account'] ?? '';
    $player  = $config['playerid'] ?? 'default';

    // Find all object tags inside play-in-place.
    $objects = $markup->xpath('//object[contains(@class, "play-in-place")]');

    if (empty($objects)) {
      return;
    }

    $replaced = FALSE;

    foreach ($objects as $object) {

      $video_ref = '';

      // Try getting ID from 'data' attribute first, then 'id'.
      if ($object->hasAttribute('data')) {
        $video_ref = $object->getAttribute('data');
      }

      if (empty($video_ref) && $object->hasAttribute('id')) {
        $video_ref = $object->getAttribute('id');
      }

      if (empty($video_ref)) {
        continue;
      }

      // Create <video-js> element.
      $video = $markup->createElement('video-js');

      $video->setAttribute('class', 'video-js');
      $video->setAttribute('controls', 'true');
      $video->setAttribute('data-video-id', $video_ref);
      $video->setAttribute('data-account', $account);
      $video->setAttribute('data-player', $player);
      $video->setAttribute('data-embed', 'default');
      $video->setAttribute('width', '640');
      $video->setAttribute('height', '360');

      // Replace object with video-js element.
      $object->parentNode->replaceChild($video, $object);
      $replaced = TRUE;
    }

    // Attach Brightcove player script via Drupal's #attached mechanism.
    if ($replaced && !empty($account) && !empty($player)) {

      $script_url = "https://players.brightcove.net/{$account}/{$player}_default/index.min.js";

      $attachment = [
        [
          '#tag' => 'script',
          '#attributes' => [
            'src' => $script_url,
            'defer' => TRUE,
          ],
        ],
        'brightcove_' . $account . '_' . $player,
      ];

      if (is_array($context)) {
        // $context is a render array — attach directly.
        $context['#attached']['html_head'][] = $attachment;
      }
      else {
        // $context is a Node entity or other object.
        // Inject the script tag directly into the markup <head> as a fallback.
        $script_element = $markup->createElement('script');
        $script_element->setAttribute('src', $script_url);
        $script_element->setAttribute('defer', 'defer');

        // Append to <head> if it exists, otherwise append to root.
        $head_elements = $markup->xpath('//head');
        if (!empty($head_elements)) {
          $head_elements[0]->appendChild($script_element);
        }
        else {
          $markup->documentElement->appendChild($script_element);
        }
      }
    }
  }

  /**
   * Config form.
   */
  public function configForm(array $config = NULL) {
    $config = $this->getConfig();

    return [
      'account' => [
        '#type' => 'textfield',
        '#title' => 'Account ID',
        '#default_value' => $config['account'] ?? '',
      ],
      'playerid' => [
        '#type' => 'textfield',
        '#title' => 'Player ID',
        '#default_value' => $config['playerid'] ?? 'default',
      ],
    ];
  }

}
